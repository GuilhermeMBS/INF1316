5) Faça um programa que leia 2 números e imprima o resultado das 4 operações básicas sobre estes 2 números.
- Verifique o que acontece se o 2º. número da entrada for 0 (zero);
- Capture o sinal de erro de floating point (SIGFPE) e repita a experiência anterior;
------------------------------------------------------------------------
ARQUIVO lab03-5.c

#include <unistd.h>
#include <sys/wait.h>
#include <signal.h>
#include <sys/ipc.h>
#include <sys/stat.h>
#include <stdlib.h>
#include <stdio.h>

#define CAPTURE_SIG 0

#if CAPTURE_SIG
void handle_sig(int signal) {
    if (signal == SIGFPE) {
        puts("Floating point error captured!");
        exit(2);
    }
}
#endif

void print_operations(int x, int y) {
    printf("%d + %d = %d\n", x, y, x + y);
    printf("%d - %d = %d\n", x, y, x - y);
    printf("%d * %d = %d\n", x, y, x * y);
    printf("%d / %d = %d\n", x, y, x / y);
}

int main(void) {
    int x, y;

    puts("Input x and y: ");
    scanf("%d %d", &x, &y);

    #if CAPTURE_SIG
    if (signal(SIGFPE, handle_sig) == SIG_ERR) {
        perror("Floating Point Signal Capture Error");
        exit(1);
    }
    #endif

    puts("\n=== Manual Inputs ===");
    print_operations(x, y);
    puts("=====================");
    puts("\n=== Case With Zero ===");
    print_operations(x, 0);
    puts("======================\n");

    return 0;
}

Por padrão, a divisão por zero gera uma falha de hardware que o Kernel traduz no sinal SIGFPE, abortando o processo com a mensagem Floating point exception (core dumped).
Ao registrar um handler para o SIGFPE, sobrescrevemos a ação default do Kernel, o que permite interceptar o erro e encerrar a execução de forma controlada via exit(2) — passo obrigatório, pois retornar do handler faria a CPU tentar reexecutar a instrução de divisão que falhou, entrando em loop infinito.

6) Faça um programa para monitorar e informar o preço de chamadas telefônicas. O programa deverá ser executado em background.
- O início e o término de uma chamada são informados através dos sinais SIGUSR1 e SIGUSR2, respectivamente.
- O custo da ligação é de 2 centavos por segundo, para ligações de até 1 minuto ou de 1 centavo por segundo a partir do 2º. minuto, ou seja, uma ligação de 1m30s custa R$1,50.
------------------------------------------------------------------------
ARQUIVO lab03-6.c

/**
 * Our program uses a time abstraction so that 1 ms represents 1 second 
 * of simulated phone call.
 */
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <time.h>
#include <sys/wait.h>
#include <sys/types.h>

// Ensures atomic access and prevents compiler register caching
typedef volatile sig_atomic_t Hflag;

static Hflag call_started = 0;
static Hflag call_ended = 0;

static struct timespec start, end;


void estimate_price(int call_id, double duration) {
    double price = duration * 2.0;
    if (duration > 60.0) price -= (duration - 60.0);
    printf("=> Price: R$ %.2f\n\n", price / 100.0);
}


void handle_child(int sig) {
    if (sig == SIGUSR1) {
        timespec_get(&start, TIME_UTC);
        call_started = 1;
    }

    else if (sig == SIGUSR2) {
        timespec_get(&end, TIME_UTC);
        call_ended = 1;
    }
}


void child_routine(int call_id) {
    signal(SIGUSR1, handle_child);
    signal(SIGUSR2, handle_child);

    while (!call_started) pause();
    printf("[Call %d] Started\n", call_id);

    while (!call_ended) pause();
    double elapsed = (end.tv_sec - start.tv_sec) * 1000.0 +
                     (end.tv_nsec - start.tv_nsec) / 1e6;
    printf("[Call %d] Ended With %.2f\n", call_id, elapsed);
    estimate_price(call_id, elapsed);

    exit(0);
}


int main(void) {
    pid_t pids[3];

    for (int i = 0; i < 3; i++) {
        pids[i] = fork();

        if (pids[i] < 0) {
            perror("Fork failed");
            exit(1);
        }

        else if (pids[i] == 0) child_routine(i + 1);
    }

    usleep(20000); // Ensure all children have registered their signal handlers
    for (int i = 0; i < 3; i++) kill(pids[i], SIGUSR1);

    // 55 seconds call (should cost around R$1,10)
    usleep(55 * 1000);
    kill(pids[1], SIGUSR2);

    // One minute and a half long call (should cost around R$1,50)
    usleep(35 * 1000); // (90 - 55) seconds
    kill(pids[0], SIGUSR2);

    // 5 minutes call (should cost around R$3,60)
    usleep(210 * 1000); // (300 - 55 - 35) seconds
    kill(pids[2], SIGUSR2);

    for (int i = 0; i < 3; i++) waitpid(pids[i], NULL, 0);

    return 0;
}

CONCLUSÃO

Para simular o monitoramento concorrente das ligações, adotamos uma arquitetura onde o processo pai atua como uma central telefônica e cada processo filho representa uma linha independente.
O pai despacha os sinais SIGUSR1 (início) e SIGUSR2 (término) individualmente para os PIDs de cada filho, garantindo que a medição dos timestamps e o cálculo de tarifação ocorram no espaço de memória isolado de cada processo, sem condições de corrida ou sobrescrita de variáveis globais.
O sincronismo interno nos filhos utiliza flags atômicas (volatile sig_atomic_t) com pause() para evitar espera ocupada e mitigar otimizações indevidas do compilador, enquanto o cálculo de tempo composto (tv_sec e tv_nsec) elimina erros decorrentes da virada de segundo.
Por fim, adotou-se uma abstração de escala onde cada milissegundo de atraso representa um segundo de chamada. Por isso, uma pequena divergência de preço (da ordem de 1 centavo) pode ocorrer de vez em quando, pois a escala está na ordem de milissegundos e o overhead acumulado das trocas de contexto, chamadas de sistema (kill, usleep), dentre outros fatores, acrescentam frações de milissegundos no processo completo.

- Elabore um programa que seja capaz de executar os 3 programas indicados anteriormente e que simule o compartilhamento da CPU entre os 3 processos com escalonamento Round-Robin com uma fatia de tempo de 1 segundo para o primeiro processo e de 2 segundos para os demais processos. Execute os programas e relate o que aconteceu
------------------------------------------------------------------------
ARQUIVO lab03-7.c

#include <unistd.h>
#include <sys/wait.h>
#include <signal.h>
#include <sys/ipc.h>
#include <sys/stat.h>
#include <time.h>
#include <stdlib.h>
#include <stdio.h>

#define QTDPROCESS 3

pid_t pids[QTDPROCESS];

void handle_sig(int signal) {
    if (signal == SIGINT) {
        puts("\nEnding child process...");
        for (int i = 0; i < QTDPROCESS; i++) kill(pids[i], SIGKILL);
        puts("Ending parent process...");
        exit(0);
    }

    return;
}


int main(void) {
    if (signal(SIGINT, handle_sig) == SIG_ERR) {
        perror("Error while starting a call.");
        exit(1);
    }

    char* sentences[] = {
        "[FILHO 1] Ola, sou o filho 1",
        "[FILHO 2] Teste do filho 2",
        "[FILHO 3] Agora o terceiro"
    };

    for (int i = 0; i < QTDPROCESS; i++) {
        pids[i] = fork();

        // Erro
        if (pids[i] < 0) {
            perror("Fork failed");
            exit(1);
        }

        // Filho
        else if (pids[i] == 0) {
            // Implementar o exec dos outros códigos
            char* args[] = {"./io1", sentences[i], NULL};
            execve("./io1", args, NULL);
            perror("Erro no execve");
            exit(1);
        }

        // Pai
        else kill(pids[i], SIGSTOP);
    }
    
    unsigned int delay;
    int atual = 0; // Processo que assume o controle3

    while (1) {
        kill(pids[atual], SIGCONT);
        delay = (atual == 0) ? 1 : 2; // 1 sec para o primeiro processo, e 2 sec para os demais 
        sleep(delay);
        kill(pids[atual], SIGSTOP);
        atual = (atual + 1) % QTDPROCESS;
    }

    return 0;
}

CONCLUSÃO

O programa coordenador atua como um escalonador preemptivo Round-Robin em espaço de usuário, controlando o ciclo de vida dos processos filhos por meio dos sinais SIGSTOP e SIGCONT.
Durante a execução, há uma alternância entre os três processos: o primeiro imprime suas mensagens durante a fatia de tempo de 1 segundo (produzindo 5 saídas com o atraso de 200 ms), enquanto os dois seguintes executam por 2 segundos cada (10 saídas cada).
Ao receber o SIGSTOP, o Kernel suspende o processo sem consumo de CPU, preservando o estado de seus registradores e buffers de I/O.
Ao receber o SIGCONT, a execução é retomada no ponto exato onde havia sido interrompida.
Ao final, o tratamento do sinal SIGINT no processo pai intercepta a interrupção do terminal (Ctrl+C) e envia SIGKILL a todos os filhos, evitando que continuem em execução em segundo plano como processos órfãos.
