#include <unistd.h>
#include <stdio.h>
#include <assert.h>

#define STEPS 3                         // Number of steps including initial and final
#define INC 8                           // Increment for second step
#define m(i) ((char*)memorypool[i])     // Used to do calculus with assert


void *memorypool[STEPS];


void*
myallocate(int n)
{
    return sbrk(n);
}


void
init_memory_pool(void)
{
    memorypool[0] = sbrk(0);
}


void
reset_memory_pool(void)
{
    /**
     * The best way to reset would be to brk(memorypool[0]), but the statement
     * explicitly asks to use the myallocate function to deallocate.
     */
    myallocate(-INC);
    memorypool[STEPS - 1] = sbrk(0);
}


static inline int
_print_memory(void)
{
    for(int i = 0; i < STEPS; i++) {
        printf("Step %d | Heap Pointer: %p\n", i, memorypool[i]);
    }

    return 1;
}


int
main(void)
{
    // Step 01: Initial Heap Pointer
    init_memory_pool();

    // Step 02: Increment of INC Bytes
    myallocate(INC);
    memorypool[1] = myallocate(0);      // Save the value in second element of array
    assert(m(1) - m(0) == INC);         // New - Initial = Increment

    // Step 03: Reset Heap Pointer
    reset_memory_pool();
    assert(m(0) == m(2));               // Initial = Final
    
    // Print the Pointers From Each Step
    assert(_print_memory() == 1);

    return 0;
}

int
main(int argc, char *argv[])
{
    struct stat sb;
    off_t len;
    char *p;
    int fd;

    if (argc < 2) {
        fprintf(stderr, "usage: %s <file>\n", argv[0]);
        exit(1);
    }

    fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror("open");
        exit(1);
    }

    if (fstat(fd, &sb)== -1) {
        perror("fstat");
        exit(1);
    }

    if (!S_ISREG(sb.st_mode)) {
        fprintf(stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }

    p = mmap(0, sb.st_size, PROT_READ, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    if (close(fd) == -1) {
        perror("close");
        exit(1);
    }

    for (len = 0; len < sb.st_size; len++) putchar(p[len]);

    if (munmap(p, sb.st_size) == -1) {
        perror("munmap");
        exit(1);
    }

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <assert.h>
#include <sys/time.h>


#define SHOW_ARCH 0


typedef enum
{
    ENV_SET = 7,
    PRINT_OK,
    MAP_OK,
    TIME_OK,
    WRITE_OK,
    DISK_OK
} ret_t;


struct timeval start, end;
struct stat sb;
off_t len;
char *p, *file_name;
int fd;


ret_t
init_environment(int argc, char *argv[])
{
    if (argc < 2) {
        fprintf(stderr, "usage: %s <file>\n", argv[0]);
        exit(EXIT_FAILURE);
    }

    fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror("open");
        exit(EXIT_FAILURE);
    }

    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        exit(EXIT_FAILURE);
    }

    if (!S_ISREG(sb.st_mode)) {
        fprintf(stderr, "%s is not a file\n", argv[1]);
        exit(EXIT_FAILURE);
    }

    file_name = argv[1];

    return ENV_SET;
}


ret_t
print_mapping_stats(void)
{
    long arch_size = sb.st_size;
    long page_size = (long)getpagesize();

    /**
     * The number of pages used is the lowest multiple of the page size
     * greater or equal to the archive size.
     * That's why we added 1 when the archive size is not a multiple of
     * the page size.
     */
    long n_pages = (arch_size / page_size) +
                   ((arch_size % page_size == 0) ? 0 : 1);

    printf("Archive Size: %ld\n", arch_size);
    printf("Page Size: %ld\n", page_size);
    printf("Number of Pages Used: %ld\n\n", n_pages);

    return PRINT_OK;
}


ret_t
calculate_elapsed_seconds(double *out_time)
{
    long seconds = end.tv_sec - start.tv_sec;
    long microseconds = end.tv_usec - start.tv_usec;

    if (microseconds < 0) {
        seconds -= 1;
        microseconds += 1000000;
    }

    double elapsed_time = seconds + (microseconds / 1000000.0);
    printf("Elapsed time: %.6f seconds\n", elapsed_time);
    *out_time = elapsed_time;

    return TIME_OK;
}


ret_t
map_archive(void)
{
    /**
     * MAP_POPULATE prefaults the page tables so that the benchmark loop
     * measures actual memory access rather than cold disk spin-up.
     */
    p = mmap(0, sb.st_size, PROT_READ, MAP_SHARED | MAP_POPULATE, fd, 0);

    if (p == MAP_FAILED) {
        perror("mmap");
        exit(EXIT_FAILURE);
    }

    if (close(fd) == -1) {
        perror("close");
        exit(EXIT_FAILURE);
    }

    gettimeofday(&start, NULL);
    for (len = 0; len < sb.st_size; len++) {
#if SHOW_ARCH
        putchar(p[len]);
#else
        volatile char dummy = p[len];
        (void)dummy;
#endif
    }
    gettimeofday(&end, NULL);

    if (munmap(p, sb.st_size) == -1) {
        perror("munmap");
        exit(EXIT_FAILURE);
    }

    return MAP_OK;
}


ret_t
write_using_disk(void)
{
    fd = open(file_name, O_RDONLY);
    if (fd == -1) {
        perror("open disk");
        exit(EXIT_FAILURE);
    }

    char ch;
    ssize_t bytes_read;

    gettimeofday(&start, NULL);
    while ((bytes_read = read(fd, &ch, 1)) > 0) {
#if SHOW_ARCH
        putchar(ch);
#else
        volatile char dummy = ch;
        (void)dummy;
#endif
    }
    gettimeofday(&end, NULL);

    if (bytes_read == -1) {
        perror("read");
        close(fd);
        exit(EXIT_FAILURE);
    }

    if (close(fd) == -1) {
        perror("close");
        exit(EXIT_FAILURE);
    }

    return DISK_OK;
}


int
main(int argc, char *argv[])
{
    double mmap_time = 0.0;
    double disk_time = 0.0;

    assert(init_environment(argc, argv) == ENV_SET);
    printf("\n=============== FILE %s ===============\n", file_name);

    assert(map_archive() == MAP_OK);
    assert(print_mapping_stats() == PRINT_OK);

    puts("Mapping Elapsed Time:");
    assert(calculate_elapsed_seconds(&mmap_time) == TIME_OK);

    puts("\nUsing I/O in Disk for the same file:");
    assert(write_using_disk() == DISK_OK);
    assert(calculate_elapsed_seconds(&disk_time) == TIME_OK);

    printf("\nSpeedup: Mapped file was %.2fx faster than standard Disk I/O.\n",
            disk_time / mmap_time);

    return 0;
}

No segundo trabalho você irá estender o seu Trabalho-1 adicionando o gerenciamento de

processo simulador do kernel/núcleo. Essas serão a implementação das tabelas de página TP,
de cada um dos cinco processos de aplicação (A1, A2, A3, A4, A5), e o mapeamento de cada
página lógica (valor da variável mem de Ax) para um dos 32 quadros de página de uma
hipotética memória principal, chamada de RAM.
Você continuará usando também o controlador de interrupções (InterContolerSim), só que ele
irá gerar, além dos IRQ0, IRQ1 e IRQ2 (como no Trabalho-1) também uma interrupção vinda
supostamente de partição de swap hipotética, digamos, a interrupção IRQ3. Esta nova
interrupção deverá sinalizar ao KernelSim quando uma operação de read/write do swap para a
memória RAM, ou vice-versa, tiver terminado. Por exemplo, para buscar uma nova página
lógica de um Ax e inseri-la em um quadro de páginas da RAM determinado pela política de
substituição de páginas a ser implementada. Ou seja, a interrupção IRQ0 continua sinalizando o
final de um TimeSlice, e as interrupções IRQ1 e IRQ2 indicam, respectivamente, o término da
operação no dispositivos D1 e D2. A interrupção IRQ3, por sua vez, indica “término de uma
operação da partição do disco de swapping”, e irá desbloquear o primeiro processo Ax na fila
de espera (pelo espaço em memória).

Novo KernelSim
O KernelSim, então, terá que ser estendido para incluir o gerenciamento de memória virtual
para cada um dos cinco processos Ax. Como a memória física RAM só tem 32 quadros de
página e são cinco A1,..A5 disputando a memória RAM, cada um com 16 páginas virtuais (16
possíveis valores da variável mem) é claro que cada Ax só terá algumas de suas páginas virtuais
posicionadas em alguns dos 32 quadros de página de RAM. Por isso o novo Kernel-Sim deverá
implementar e atualizar, constantemente as seguintes estruturas de dados:
● globalmente, um único array, RAM [32], representando a memória física RAM com 32
page frames e
● um array RAM-free[32] de booleanos para designar os frames livres em RAM[32]
● uma tabela de página, TPx[16], uma para cada processo Ax, indicando quais de suas
páginas virtuais estão em qual índice de pageframe no array de RAM.
● Cada entrada de uma tabela de página TP[i] deverá ter vários campos, mas pelo menos
os seguintes:
○ valid/invalid bit: se essa página está na memória RAM ou não (valid =0)
○ frame: índice (ou ponteiro) para o page frame onde está localizado na RAM
○ modifyBit: se a última operação que ocasionou a obtenção de um page frame
da RAM foi uma escrita, então modifyBit=1, senão modifyBit=0
○ when: o valor do PC de Ax na ocasião do mais recente access(mem,op)
quando mem conseguiu de um page frame da RAM
Passo a passo da tradução de memória lógica para física
2
Então, ao receber um access(mem,op)
2 de um processo Ax, o KernelSim:
1. Bloqueia o processo Ax (SIGSTOP)
2. Verifica na tabela de páginas TPx[mem] se a entrada mem está válida, isto é, contém
um índice/endereço de quadro de página na memória física RAM, ou não.
3. Se a página mem estiver na RAM (bit válido), imediatamente desbloqueia Ax
(SIGCONT) e coloca na fila de processos prontos;
4. Se mem não estiver na RAM, temos um Page-faul e faça:
a. Procure, em RAM-free, se existe um quadro de página dando sopa (= vazio). E
se sim, atribui a TPx[mem].frame o endereço do quadro de páginas escolhido,
desbloqueia Ax (SIGCONT);
b. Se todos os quadros da RAM estiverem ocupados por páginas de processos,
chame o seu algoritmo de substituição de página substitute(). Este retornará
um quadro da RAM a ser alocado para Ax.mem e a ser sobrescrito, digamos,
RAM[zz].
c. Atualize então a tabela TPx[mem].frame com zz, o endereço do quadro de
página retornado por subs() e insira a requisição (Ax,mem) no final da fila de
espera (Sw_Queue), aguardando por uma interrupção IRQ3 do disco de swap;
d. Então, quando essa requisição chegar ao início da Sw_Queue, é como se o
conteúdo da página mem tivesse sido lido da partição de swap e possa ser
escrito na RAM[zz];
e. Mas existem dois casos:
i. se o conteúdo prévio de RAM[zz] tiver sido uma página apenas lida,
devido à operação prévia de access(..,R), o recebimento de uma única
IRQ3 basta para sobrescrever o quadro RAM[zz] com a página Ax.mem.
ii. Porém, se conteúdo prévio de RAM[zz] tiver sido uma página escrita,
este conteúdo prévio teria que ser previamente atualizado no disco de
swapping. Ou seja, demandaria, primeiro, a escrita do conteúdo
anterior no swap e, só depois, a leitura da página Ax.mem do swap, ou

CONCLUSÃO

Quando começamos a medir o tempo do mmap(), percebemos logo de cara que cronometrar apenas a chamada da função era um erro, porque o SO usa carregamento sob demanda (lazy loading): a função só registra o ponteiro na memória virtual, mas o arquivo só sai do disco para a RAM quando o nosso laço for tenta ler os endereços e força os Page Faults na MMU.
Ao colocar o cronômetro no laço de leitura, notamos que o putchar() deixava tudo lento por culpa do tempo de resposta visual do terminal.
Para isolar a velocidade do hardware, desativamos a impressão e usamos uma variável volatile para forçar a CPU a ler byte por byte na memória.
Já na comparação com a E/S tradicional usando read(), descartamos a ideia de criar um buffer de 4096 bytes (1 página), pois o read() estava dando tempos menores que o mmap().
Analisando os logs, descobrimos que como o mmap() rodava primeiro, ele fazia todo o trabalho pesado de buscar o arquivo "frio" no SSD e sofrer as interrupções da MMU, deixando o arquivo salvo no cache da RAM (Page Cache do Kernel) para o read() ler quase instantaneamente logo depois.
Por isso nivelamos o teste usando a flag MAP_POPULATE no mmap (para pré-carregar as páginas) e comparamos a leitura direta byte a byte.
Dessa forma a vantagem real apareceu: enquanto o read() perde muito tempo fazendo chamadas de sistema (syscalls), trocando contexto entre Usuário e Kernel e copiando dados para um buffer intermediário, o mmap() delega todo o controle de blocos de 4 KB diretamente para o hardware (MMU e TLB), permitindo varrer o arquivo como um simples vetor na velocidade da própria RAM.

ARQUIVO prog1.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>


#define FILE_NAME "shared_data.bin"
#define SHARED_SIZE 4096


int
main(void)
{
    int fd = open(FILE_NAME, O_RDWR | O_CREAT, 0666);
    if (fd == -1) {
        perror("open prog1");
        exit(EXIT_FAILURE);
    }

    // Defines the archive's size (extends it to avoid Seg Fault)
    if (ftruncate(fd, SHARED_SIZE) == -1) {
        perror("ftruncate");
        close(fd);
        exit(EXIT_FAILURE);
    }

    char *shared_mem = mmap(NULL, SHARED_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (shared_mem == MAP_FAILED) {
        perror("mmap prog1");
        close(fd);
        exit(EXIT_FAILURE);
    }
    
    close(fd);

    printf("PROG 1 | Writing in shared memory...\n");
    const char *msg = "[PROG 1] Message from Prog1. Can you read me?";
    strncpy(shared_mem, msg, SHARED_SIZE);
    printf("PROG 1 | Message successfully written!\n");

    munmap(shared_mem, SHARED_SIZE);
    
    return 0;
}

5) Escreva um programa que faz um mapeamento de arquivo para memória e depois escreve novos caracteres no final do arquivo. O que voce precisa fazer para evitar um SegFault?
O Linux também suporta uma função:

    new_sz, unsigned long flags) // addr precisa estar alinhado com o início da página

ARQUIVO q5.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/stat.h>


#define FILE_NAME "data.txt"


int
main(void)
{
    int fd = open(FILE_NAME, O_RDWR | O_CREAT, 0666);
    if (fd == -1) {
        perror("open");
        exit(EXIT_FAILURE);
    }

    struct stat sb;
    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        close(fd);
        exit(EXIT_FAILURE);
    }

    size_t prev_size = sb.st_size;
    const char *new_text = "==> Extension Text\n";
    size_t new_size = prev_size + strlen(new_text);

    // Extends the archive's size, preventing Seg Fault
    if (ftruncate(fd, new_size) == -1) {
        perror("ftruncate");
        close(fd);
        exit(EXIT_FAILURE);
    }

    char *p = mmap(NULL, new_size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        close(fd);
        exit(EXIT_FAILURE);
    }
    
    close(fd);

    strcpy(&p[prev_size], new_text);
    puts("\nNew text successfully added!");
    printf("Previous Size: %zu bytes\nNew Size: %zu bytes.\n", 
           prev_size, new_size);

    munmap(p, new_size);

    return 0;
}

6) Re-escreva o seu programa do item anterior usando mremap()

ARQUIVO q6.c

#define _GNU_SOURCE // Needed to use the flag MREMAP_MAYMOVE in mremap()


#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/stat.h>


#define FILE_NAME "data.txt"


int
main(void)
{
    int fd = open(FILE_NAME, O_RDWR | O_CREAT, 0666);
    if (fd == -1) {
        perror("open");
        exit(EXIT_FAILURE);
    }

    struct stat sb;
    if (fstat(fd, &sb) == -1) {
        perror("fstat");
        close(fd);
        exit(EXIT_FAILURE);
    }

    /**
     * Separate the actual file size (write offset) from the initial mapping size.
     * If the file is empty (0 bytes), we map 1 dummy byte to satisfy mmap(),
     * but keep write_offset at 0 so we don't leave a leading NULL byte at index 0.
     */
    size_t write_offset = (size_t)sb.st_size;
    size_t old_map_size = (write_offset > 0) ? write_offset : 1;

    char *p = mmap(NULL, old_map_size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        close(fd);
        exit(EXIT_FAILURE);
    }

    const char *new_text = "==> Text appended using mremap.\n";
    size_t text_len = strlen(new_text);
    size_t new_size = write_offset + text_len;

    if (ftruncate(fd, (off_t)new_size) == -1) {
        perror("ftruncate");
        munmap(p, old_map_size);
        close(fd);
        exit(EXIT_FAILURE);
    }

    char *new_p = mremap(p, old_map_size, new_size, MREMAP_MAYMOVE);
    if (new_p == MAP_FAILED) {
        perror("mremap");
        munmap(p, old_map_size);
        close(fd);
        exit(EXIT_FAILURE);
    }

    // Write starting at the true end of the original file (0 if initially empty)
    memcpy(&new_p[write_offset], new_text, text_len);

    printf("Mapping resized from %zu to %zu bytes at address %p\n",
           write_offset, new_size, (void *)new_p);

    if (munmap(new_p, new_size) == -1) {
        perror("munmap");
        close(fd);
        exit(EXIT_FAILURE);
    }

    if (close(fd) == -1) {
        perror("close");
        exit(EXIT_FAILURE);
    }

    return EXIT_SUCCESS;
}

Nesta questão, reescrevemos o programa para expandir o mapeamento dinamicamente usando mremap(), funcionando como um realloc() para páginas de memória sem precisar desmapear e mapear tudo de novo.
Primeiro esticamos o arquivo físico no disco com ftruncate() e, em seguida, expandimos a área na memória virtual com mremap() usando a flag MREMAP_MAYMOVE (que permite ao Kernel mover os ponteiros na Tabela de Páginas caso não haja espaço contíguo livre).
Durante os testes, notamos que quando o arquivo nascia vazio (0 bytes), precisávamos forçar o tamanho inicial do mmap para 1 byte para a chamada não falhar.
Porém, como o ftruncate preenche o espaço novo com zeros, usar esse tamanho "falso" de 1 byte como índice de escrita acabava pulando a posição 0 e deixava um caractere NULL (\0) no começo do arquivo.
Resolvemos isso separando o tamanho mínimo do mapeamento (old_map_size) do deslocamento real de escrita (write_offset = 0) e trocando strcpy por memcpy, gravando o texto na posição exata sem caracteres nulos indesejados.
