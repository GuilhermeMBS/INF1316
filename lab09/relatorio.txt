1) Use o programa listaarquivo.c para imprimir o numero de inode, o tamanho (em bytes) e a idade em dias de cada arquivo do diretório corrente.
Lembre-se que stat() espera o path completo do arquivo.
------------------------------------------------------------------------
ARQUIVO listaarquivo.c

#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <unistd.h>

#include <sys/types.h>
#include <sys/stat.h>
#include <sys/dir.h>
#include <sys/param.h>

char pathName[MAXPATHLEN];

extern int alphasort();

int main() {
    int count;
    struct direct **files;

    if (getwd(pathName) == NULL) perror("Error while opening current dir");

    count = scandir(pathName, &files, NULL, alphasort);
    if (count <= 0) perror("No files in this directory");

    printf("Number of files = %d\n", count);

    for (int i = 0; i < count; i++) {
        struct stat atributes;

        if (stat(files[i]->d_name, &atributes) == 0) {
            time_t now = time(NULL);

            double time_passed = difftime(now, atributes.st_ctime);
            int days_passed = time_passed / (60 * 60 * 24);

            printf("%s \t inode %lu \t size: %ld \t age: %d\n", 
                files[i]->d_name, 
                (unsigned long)atributes.st_ino, 
                atributes.st_size, 
                days_passed
            );
        }
    }

    return 0;
}

3) Escreva um programa que percorre recursivamente um diretório, a partir do diretório corrente, somando o tamanho de todos os arquivos encontrados.
------------------------------------------------------------------------
ARQUIVO teste/ola.txt

idwahnubiua
wdahbdwyiuag8yui

ARQUIVO listaSubDir.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

#include <sys/types.h>
#include <sys/stat.h>
#include <sys/dir.h>
#include <sys/param.h>
#include <dirent.h>

char pathName[MAXPATHLEN];

extern int alphasort();

long long sumSubDir(const char *dirName) {
    DIR *dir;
    struct dirent *entry;
    struct stat atributes;
    
    long long totalSize = 0;
    char path[MAXPATHLEN];

    dir = opendir(dirName);
    if (dir == NULL) perror("Error while opening the directory");

    while ((entry = readdir(dir)) != NULL) {
        if (strcmp(".", entry->d_name) == 0 || strcmp("..", entry->d_name) == 0) continue;
        
        snprintf(path, sizeof(path), "%s/%s", dirName, entry->d_name);
        
        if (stat(path, &atributes) == -1) perror("Error while opening the file");
        
        if (entry->d_type == DT_DIR) totalSize += sumSubDir(path);
        else if (entry->d_type == DT_REG) totalSize += atributes.st_size;
    }

    closedir(dir);
    return totalSize;
}

int main() {
    printf("Soma dos tamanhos de todos os arquivos: %lld\n", sumSubDir("."));

    return 0;
}

ARQUIVO listasubdir.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

#include <sys/types.h>
#include <sys/stat.h>
#include <sys/dir.h>
#include <sys/param.h>
#include <dirent.h>

char pathName[MAXPATHLEN];

extern int alphasort();

long long sumSubDir(const char *dirName, int level) {
    DIR *dir;
    struct dirent *entry;
    struct stat atributes;
    
    long long totalSize = 0;
    char path[MAXPATHLEN];

    dir = opendir(dirName);
    if (dir == NULL) perror("Error while opening the directory");

    while ((entry = readdir(dir)) != NULL) {
        if (strcmp(".", entry->d_name) == 0 || strcmp("..", entry->d_name) == 0) continue;
        
        snprintf(path, sizeof(path), "%s/%s", dirName, entry->d_name);
        
        if (stat(path, &atributes) == -1) perror("Error while opening the file");
        
        printf("%*s[%s]\n",
            level * 4,
            "",
            entry->d_name
        );
        
        if (entry->d_type == DT_DIR) totalSize += sumSubDir(path, level + 1);
        else if (entry->d_type == DT_REG) totalSize += atributes.st_size;
    }

    closedir(dir);
    return totalSize;
}

int main() {
    printf("[root]\n");
    long long total = sumSubDir(".", 1);

    printf("Soma dos tamanhos de todos os arquivos: %lld\n", total);

    return 0;
}
